\subsection{Background}

\begin{frame}{Malthusian Growth Model}
\begin{columns}
    \begin{column}{0.6\textwidth}
        \begin{itemize}
    \item The Malthusian growth model is a mathematical model of population growth.
   \item It essentially describes exponential population growth. [Image of exponential growth curve]
   \item The model is named after Thomas Malthus, who wrote \textit{An Essay on the Principle of Population} (1798), one of the first and most influential books on population dynamics.
  \end{itemize}
    \end{column}

    \begin{column}{0.4\textwidth}
    \includegraphics[scale = 0.08]
    {images/malthus_portrait.jpg}
       Thomas Robert Malthus (1766 - 1834), an English economist and scholar.
    \end{column}
\end{columns}
\end{frame}
  
\begin{frame}{Key assumptions of Thomas Malthus}
\begin{itemize}
    \item \textbf{Population grows exponentially when resources are abundant.}
    \item Malthus further assumed that agricultural production increases only linearly (arithmetically). [Image of Malthusian catastrophe graph showing exponential population vs linear resources]
    \item Population growth is limited by available resources and can be reduced by famine, disease, or war.
\end{itemize}
\vfill
\small

%\pause
\begin{block}{Malthusian Catastrophe}
A situation where the population exceeds the agricultural production of its environment. This results in severe consequences like poverty, famine, disease, and war. Malthus believed a Malthusian catastrophe was inevitable unless population growth was controlled.
\end{block}
\end{frame}


\begin{frame}{Malthus, Darwin, and Natural Selection}
    \begin{itemize}
        \item Before reading Malthus, Darwin thought that living things reproduced just enough individuals to keep populations stable.
        \item Malthus helped him realize that as populations breed beyond their means, advantageous traits become more prevalent due to competition for resources.
        \item This led to the development of the theory of natural selection.
\end{itemize}
\vfill

\begin{frame}{Neo-Malthusianism}
\small
\begin{columns}
    \begin{column}{0.7\textwidth}
        \begin{itemize}
    \item A revival of Malthusian concerns in the 20\textsuperscript{th} century, often focused on environmental limits.
\item Neo-Malthusians advocate for population control due to concerns about overpopulation, resource depletion, and environmental degradation.
\item Widely discussed in the 1960s; a famous example is Paul Ehrlich's book \textit{The Population Bomb} (1968).
\item Their influence can be seen in family planning initiatives and advocacy for sustainability.
\item Modern environmentalism acknowledges the impact of population growth on resource use, pollution, and environmental strain.
\end{itemize}

\begin{frame}{Malthusian Growth Model Equation}

The change in population $dP/dt$ is proportional to the current population size $P$ at time $t$. The parameter $r$ is the intrinsic rate of increase (the difference between birth rate and death rate).
    \begin{align*}
    \frac{dP}{dt} &= rP
  \end{align*}
\vfill
%\pause

To solve the differential equation, we separate the variables.
We move terms with $P$ to one side and terms with $t$ to the other side.
    \begin{align*}
    \frac{dP}{P} &= r dt
  \end{align*}
\end{frame}


\begin{frame}{Malthusian Growth Model Equation}

We integrate both sides of the equation. Integration introduces a constant of integration, $C$.
   \begin{align*}
    \int \frac{dP}{P} &= \int r dt \\
    \ln|P| &= rt + C
  \end{align*}

%\pause
We exponentiate both sides. We remove the absolute value as the population size is always positive.
\begin{align*}
e^{\ln P} &= e^{rt + C}  \\
P &= e^{C} e^{rt}  
\end{align*}
%\pause
The integration constant $e^C$ represents the initial population $P_0$. Substitution leads to the final explicit form:
\begin{align*}
    P(t) &= P_0 e^{rt} 
  \end{align*}
\end{frame}

\begin{frame}[t]{Numerical vs Explicit Solution}
  
\begin{itemize}
    \item Numerical and explicit solutions are two methods used to solve mathematical problems in modeling and simulation.
        
    \vspace{1em}
    
    \only<1>{
    \item \textbf{Explicit (Analytical) Solution}: A mathematically \textbf{exact} solution expressed as a formula.
    \begin{itemize}
    \color{positive}
        \item Provides a precise value for any time $t$.
        \item Often allows for a deeper understanding of the underlying mathematical relationships.
    \color{negative}
        \item It can be very difficult or impossible to find explicit solutions for complex, non-linear problems.
    \end{itemize}
    }
    \only<2>{
    \item \textbf{Numerical Solution}: Uses \textbf{iterative methods to approximate} a solution using computational steps. [Image of Euler method steps visualization]

       \begin{itemize}
        \color{positive}
        \item Allows solving a much wider range of complex problems (e.g., where parameters change over time).
        \color{negative}
        \item Solutions are only approximate.
        \item Solutions can be affected by cumulative calculation errors (discretization error).
        \item Difficult to validate without real-world data or an explicit solution for comparison.
    \end{itemize}}
\end{itemize}
\end{frame}

\begin{frame}{Numerical vs Explicit Solution for Malthusian Growth}
\footnotesize

\begin{columns}
    \begin{column}{0.45\textwidth}
       \textbf{Explicit Solution}
       
       Calculation using the formula:
       \begin{equation*}
    P(t) = P_0 e^{rt} 
  \end{equation*}

  Where:
\begin{itemize}
    \item $P_0$ is the initial population at time $0$
    \item $r$ is the growth rate
    \item $t$ is the time
\end{itemize}

        
    \end{column}
    
    %\pause
  \hspace{1em}
\vrule{}
  \hspace{1em}

    \begin{column}{0.45\textwidth}
   \textbf{Numerical Solution (Euler's Method):}  
   \begin{equation*}
   \frac{dP}{dt} \approx \frac{\Delta P}{\Delta t}
\end{equation*}

\vspace{1em}
  Algorithm:
 \begin{enumerate}
     \item \textbf{Discretization:}
     \begin{equation*}
         t_{i+1} = t_i + \Delta t
     \end{equation*}
        \item \textbf{Iteration (Update Rule):}
     \begin{equation*}
         P(t_{i+1}) = P(t_i) + r P(t_i) \Delta t
     \end{equation*}
 \end{enumerate}
Where:
\begin{itemize}
    \item $P(t_i)$ is the population at time step $i$
    \item $\Delta t$ is the time step size
\end{itemize}
    \end{column}
\end{columns}
\end{frame}
