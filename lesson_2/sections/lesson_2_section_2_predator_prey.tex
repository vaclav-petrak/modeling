\begin{frame}
    \frametitle{Overview}
    \begin{itemize}
        \item The Lotka-Volterra equations, also known as the predator-prey equations, model the dynamics of biological systems in which two species interact, one as a \textbf{predator} and the other as \textbf{prey}.
        \item They were proposed independently by Alfred J. Lotka in 1925 and Vito Volterra in 1926.
        \item   Model is characterized by \textbf{oscillations} in the population size of both predator and prey,
        \item Peak of the predator's oscillation lags  behind the peak of the prey's oscillation.
    \end{itemize}
\end{frame}

\begin{frame}{Assumptions}
         The model makes several simplifying assumptions but captures the essential dynamics of predator-prey interactions:
        \begin{itemize}
         \pause
          \item  the prey population will grow exponentially when the predator is absent (unrealistic in a long term but may be precise for a long period) 
           \pause
          \item the predator population will starve in the absence of the prey population --- there is no option of  switching to another type of prey (may be realistic for some circumstances) 
           \pause
          \item predators can consume any quantities of prey (unrealistic for large quantities)
           \pause
          \item both populations are moving randomly through a homogeneous, static, environment (may be approximately true for some environments)
        \end{itemize}

\end{frame}

\begin{frame}{The Mathematical Model}
\small
    The Lotka-Volterra equations are a pair of first-order, nonlinear, differential equations given by:
    \begin{equation}
        \frac{dN}{dt} = rN - \beta NP
    \end{equation}
      \begin{equation}
        \frac{dP}{dt} = m\beta NP - dP
    \end{equation}
    \pause
    where:
    \begin{itemize}
        \item $N$ is population number of prey (e.g., rabbits),
            \pause
        \item $P$ is the number of predators (e.g., foxes),
        \item $r$ is the prey growth rate
            \pause
        \item $d$ is the death rate of predators
            \pause
        \item $m$ is the proportionality constant
            \pause
        \item $\beta$ attack efficiency of the predator
    \end{itemize}
\end{frame}


\begin{frame}{Prey equations}
\footnotesize
    The prey population is given by:
    \begin{equation}
        \frac{dN}{dt} = rN - \beta NP
    \end{equation}
    
    \pause
    In the absence of predators, the equation becomes:
    \begin{equation}
        \frac{dN}{dt} = rN 
    \end{equation}
    which represents the Malthusian model. The prey population would grow exponentially with the growth rate $r$.

    \pause
 \vfill{1em}
    The reduction in prey population due to predators is captured by the expression:
    \begin{equation}
        - \beta NP
    \end{equation}
    \pause
    Here, $NP$ is proportional to the number of encounters. An increased number of predators, an increased prey population, and greater searching efficiency of the predators would all lead to a higher removal rate of prey.
    $\beta$ is attack efficiency of the predator. Higher $\beta$ means higher probability of prey being killed.
\end{frame}

\begin{frame}{Predator equations}
\footnotesize
    The predator population is given by:
      \begin{equation}
        \frac{dP}{dt} = m\beta NP - dP
    \end{equation}
    \pause
    If the prey population $N$ is 0, the equation becomes:
    \begin{equation}
        \frac{dN}{dt} = - dN 
    \end{equation}
    which represents dying of prey at exponential rate $d$.

\vfill
     \pause
    The growth of predator population growth by eating of prey
    \begin{equation}
        + m \beta NP
    \end{equation}
    Here we see again $\beta NP$. Proportionality constant $m$ tells how many prey individuals are required for increase of predator population by one. 
\end{frame}

\begin{frame}{What is a Vector Field?}
    \begin{itemize}
        \item In mathematics and physics, a \textbf{vector field} assigns a vector to every point in a given space. 
        \pause
        \item For a 2D dynamical system, the state is represented by coordinates $(x, y)$. The differential equations define the rate of change at any point:
        \begin{equation}
            \vec{v}(x,y) = \left( \frac{dx}{dt}, \frac{dy}{dt} \right)
        \end{equation}
        \pause
        \item Visually, this is represented as a grid of arrows. The \textbf{direction} of the arrow shows where the system is heading, and its \textbf{length} (magnitude) shows how fast it is changing.
        \item Vector fields allow us to understand the qualitative, long-term behavior of a system without needing to analytically solve the complex differential equations.
    \end{itemize}
\end{frame}

\begin{frame}{Vector Field in Predator-Prey Phase Space}
    \begin{itemize}
        \item In the Lotka-Volterra model, our 2D phase space is defined by the prey population $N$ and the predator population $P$.
        \pause
        \item At any given state $(N, P)$, the instantaneous rate of change is the vector:
        \begin{equation}
            \vec{v} = \left( rN - \beta NP, \; m\beta NP - dP \right)
        \end{equation}
        \pause
        \item The arrows in this specific vector field indicate how both populations will shift from that exact moment.
        \item By following the continuous flow of these arrows, we can trace the closed, periodic trajectories of the populations over time.
    \end{itemize}
\end{frame}

\begin{frame}{Calculating the Vector Field: Examples}
    \begin{itemize}
        \small
        \item Let's calculate the instantaneous rate of change $\vec{v} = \left(\frac{dN}{dt}, \frac{dP}{dt}\right)$ using a specific set of parameters: 
        \item \textbf{Assume:} $r = 2$ (prey growth), $\beta = 0.1$ (attack rate), $m = 0.5$ (conversion), and $d = 1$ (predator death). 
        \item Our rate equations become:
        \begin{equation}
            \frac{dN}{dt} = 2N - 0.1NP \quad \text{and} \quad \frac{dP}{dt} = 0.05NP - P
        \end{equation}
        
        \vspace{0.5em}
        
        \only<2>{
            \item \textcolor{blue}{\textbf{Point 1: (10, 10) --- Low prey, low predators}}
            \begin{equation*}
                 \textcolor{blue}{\vec{v} = (2(10) - 0.1(10 \cdot 10), \; 0.05(10 \cdot 10) - 10) = \mathbf{(10, -5)}} 
            \end{equation*}
            \textit{\textcolor{blue}{Result: Prey population grows (+10), predators starve (-5).}}
        }
        
        \only<3>{
            \item \textcolor{red}{\textbf{Point 2: (40, 10) --- Abundant prey, low predators}}
            \begin{equation*}
                 \textcolor{red}{\vec{v} = (2(40) - 0.1(40 \cdot 10), \; 0.05(40 \cdot 10) - 10) = \mathbf{(40, 10)}} 
            \end{equation*}
             \textit{\textcolor{red}{Result: Both populations are booming.}}
        }
        
        \only<4>{
            \item \textcolor{green!60!black}{\textbf{Point 3: (20, 20) --- The Coexistence Point}}
            \begin{equation*}
                 \textcolor{green!60!black}{\vec{v} = (2(20) - 0.1(20 \cdot 20), \; 0.05(20 \cdot 20) - 20) = \mathbf{(0, 0)}} 
            \end{equation*}
             \textit{\textcolor{green!60!black}{Result: The system is perfectly balanced; no change.}}
        }
    \end{itemize}
\end{frame}

\begin{frame}{Equilibrium Points in the Vector Field}
    \begin{itemize}
        \item \textbf{Equilibrium points} (or fixed points) occur where the vector magnitude is exactly zero. At these points, the system is completely at rest ($\frac{dN}{dt} = 0$ and $\frac{dP}{dt} = 0$).
        \pause
        \item For the Lotka-Volterra equations, the vector field reveals two such points:
        \begin{enumerate}
            \item \textbf{Extinction:} $(0, 0)$ --- both populations are zero.
            \pause
            \item \textbf{Coexistence:} $\left( \frac{d}{m\beta}, \frac{r}{\beta} \right)$ --- populations perfectly balance each other out and remain static.
        \end{enumerate}
        \pause
        \vspace{1em}
        \item The vector field arrows orbit continuously around the coexistence equilibrium point, mathematically illustrating the perpetual lagging and catching-up dynamics of the two species.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Biological Implications}
    \begin{itemize}
        \item The model predicts oscillations in the populations of predator and prey.
        \item The fluctuation of the two species is periodic,
    \item The amplitude and period of these oscillations depend on the initial conditions and the parameters of the model.
    \item  The \textit{average} numbers of the two species are constant
    \item  If we  destroy individuals of both species  proportionately to their number, the average number of individuals of the prey grows and the average number of predators diminishes.
    \end{itemize}
\end{frame}



\begin{frame}
    \frametitle{Conclusion}
    \begin{itemize}
        \item The Lotka-Volterra predator-prey model provides a  framework for understanding the dynamics of biological systems with predator-prey interactions.
        \item Despite its simplicity, it offers valuable insights into the oscillatory nature of ecological populations and has motivates further research into more complex and realistic models.
    \end{itemize}
\end{frame}

\begin{frame}{Calculating Equilibrium Points: Step 1}
    \begin{itemize}
        \item To find the exact equilibrium points, we must determine mathematically where the populations stop changing.
        \pause
        \item This requires setting the rates of change (derivatives) for both populations to zero:
        \begin{equation}
            \frac{dN}{dt} = 0 \quad \text{and} \quad \frac{dP}{dt} = 0
        \end{equation}
        \pause
        \item Substituting our Lotka-Volterra equations, we establish a system of two algebraic equations:
        \begin{equation}
            rN - \beta NP = 0
        \end{equation}
        \begin{equation}
            m\beta NP - dP = 0
        \end{equation}
    \end{itemize}
\end{frame}

\begin{frame}{Calculating Equilibrium Points: Step 2}
    
    \begin{itemize}
        \item Next, we factor $N$ and $P$ out of both equations to simplify solving them:
        \begin{equation}
            N(r - \beta P) = 0
        \end{equation}
        \begin{equation}
            P(m\beta N - d) = 0
        \end{equation}
        \pause
        \vspace{1em}
        \item \textbf{Case 1: The Extinction Equilibrium}
        \item From the first equation, one obvious solution is $N = 0$.
        \pause
        \item Substituting $N = 0$ into the second equation gives $P(-d) = 0$. Since the death rate $d > 0$, this means $P$ must be $0$.
        \item This gives us our first equilibrium point at the origin: \textbf{$(0, 0)$}. 
    \end{itemize}
\end{frame}

\begin{frame}{Calculating Equilibrium Points: Step 3}
    \begin{itemize}
        \item \textbf{Case 2: The Coexistence Equilibrium}
        \item Now, let's assume neither population is extinct ($N \neq 0$ and $P \neq 0$). 
        \pause
        \item To satisfy our factored equations, the terms inside the parentheses must equal zero:
        \begin{equation}
            r - \beta P = 0
        \end{equation}
        \begin{equation}
            m\beta N - d = 0
        \end{equation}
        \pause
        \item Solving these for $P$ and $N$ respectively:
        \begin{equation}
            P = \frac{r}{\beta} \quad \text{and} \quad N = \frac{d}{m\beta}
        \end{equation}
        \pause
        \item This yields our second, non-trivial equilibrium point: \textbf{$\left( \frac{d}{m\beta}, \frac{r}{\beta} \right)$}.
    \end{itemize}
\end{frame}

\section{Modeling Approaches}

\begin{frame}{Modeling Approach 1: The Euler Method}
    \begin{itemize}
        \item The Euler method is the simplest numerical procedure for solving ordinary differential equations (ODEs).
        \vspace{0.5em}
        \item \textbf{How it works:} We break time into discrete, tiny steps ($\Delta t$). At each step, we calculate the current rate of change and assume it remains constant for that brief interval.
        \item \textbf{Discrete Equations:}
        \begin{equation}
            N_{t+\Delta t} = N_t + \Delta t (r N_t - \beta N_t P_t)
        \end{equation}
        \begin{equation}
            P_{t+\Delta t} = P_t + \Delta t (m \beta N_t P_t - d P_t)
        \end{equation}
        \vspace{0.5em}
        \item \textbf{Pros:} Transparent; excellent for learning how continuous equations translate to discrete steps.
        \item \textbf{Cons:} Prone to accumulating errors. If $\Delta t$ is too large, the inherent closed loops artificially spiral outwards.
    \end{itemize}
\end{frame}

\begin{frame}[fragile]{Euler Method in Python}
    \footnotesize
    Demonstrating the explicit loop, mapping directly to the discrete equations:
    \vspace{0.5em}
    \begin{lstlisting}[language=Python, basicstyle=\ttfamily\scriptsize, keywordstyle=\color{blue}, commentstyle=\color{gray}, frame=single]
import numpy as np

r, beta, m, d = 1.0, 0.1, 0.05, 0.5
dt = 0.01  # Small time step is crucial for stability
steps = 2000

N, P = np.zeros(steps), np.zeros(steps)
N[0], P[0] = 40, 9  # Initial conditions

for i in range(1, steps):
    dN_dt = r * N[i-1] - beta * N[i-1] * P[i-1]
    dP_dt = m * beta * N[i-1] * P[i-1] - d * P[i-1]
    
    N[i] = N[i-1] + dN_dt * dt
    P[i] = P[i-1] + dP_dt * dt
    \end{lstlisting}
\end{frame}

\begin{frame}{Modeling Approach 2: ODE Solvers}
    \begin{itemize}
        \item For practical applications and research, manual loops are often insufficient. We use dedicated mathematical solvers, such as SciPy's \texttt{solve\_ivp}.
        \vspace{0.5em}
        \item \textbf{How it works:} These solvers act as a "black box" utilizing advanced algorithms (like Runge-Kutta). You provide the differential equations, time span, and initial conditions.
        \item \textbf{Adaptive Step Size:} The solver automatically shrinks $\Delta t$ when populations change rapidly (for accuracy) and expands it when stable (to save computing power).
        \vspace{0.5em}
        \item \textbf{Pros:} Highly accurate, stable, and correctly handles the periodic nature of predator-prey dynamics without artificial spiraling.
        \item \textbf{Cons:} Less transparent for beginners; the integration loop is hidden behind the scenes.
    \end{itemize}
\end{frame}

\begin{frame}[fragile]{SciPy Solver in Python}
    \footnotesize
    Define the system as a function, allowing the solver to handle integration:
    \vspace{0.5em}
    \begin{lstlisting}[language=Python, basicstyle=\ttfamily\scriptsize, keywordstyle=\color{blue}, commentstyle=\color{gray}, frame=single]
from scipy.integrate import solve_ivp

def lotka_volterra(t, state, r, beta, m, d):
    N, P = state
    dN_dt = r * N - beta * N * P
    dP_dt = m * beta * N * P - d * P
    return [dN_dt, dP_dt]

params = (1.0, 0.1, 0.05, 0.5)
initial_state = [40, 9] 
t_span = (0, 20) 

# 'RK45' is the default Runge-Kutta method
solution = solve_ivp(fun=lotka_volterra, t_span=t_span, 
                     y0=initial_state, args=params)

time = solution.t
prey, predators = solution.y[0], solution.y[1]
    \end{lstlisting}
\end{frame}
